def _load_high_score(self):
    try:
        with open('highscore.txt') as f:
            return int(f.read().strip() or 0)
    except (OSError, ValueError):
        return 0
def _save_high_score(self):
    try:
        with open('highscore.txt', 'w') as f:
            f.write(str(self.high_score))
    except OSError:
        pass
